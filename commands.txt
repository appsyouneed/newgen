#!/bin/bash
# Newgen Service Commands

## Setup (run once from /root/termux/newgen/)
sudo bash setup.sh

## Start
sudo systemctl start newgen

## Stop
sudo systemctl stop newgen

## Restart
sudo systemctl restart newgen

## Status
sudo systemctl status newgen

## Live logs
sudo journalctl -u newgen -f

## Last 100 lines
sudo journalctl -u newgen -n 100

## Kill manually
sudo pkill -f "python3.*app.py"

## Switch to newgen from vidgen
sudo systemctl stop vidgen && sudo systemctl disable vidgen
sudo systemctl enable newgen && sudo systemctl start newgen

## Switch back to vidgen
sudo systemctl stop newgen && sudo systemctl disable newgen
sudo systemctl enable vidgen && sudo systemctl start vidgen
