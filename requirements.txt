# --- HuggingFace stack -------------------------------------------------------
# huggingface-hub must stay <1.0 for transformers<5.
# gradio 4.43.0 imports HfFolder which was removed in huggingface-hub>=0.26.
# setup.sh patches gradio/oauth.py with a shim after install so this is safe.
huggingface-hub>=0.34.0,<1.0

# transformers PINNED EXACTLY — do not turn this back into a range.
# Needs >=4.52.0 for Qwen-Image-Edit-2511's nested text_config format
# (Qwen2_5_VLConfig.sub_configs lacked "text_config" before 4.52.0, so the
# nested dict stayed raw and GenerationConfig.from_model_config() crashed
# calling .to_dict() on it).
# transformers 4.57.6 was tested here and BREAKS diffusers' Wan pipeline:
# its lazy-import machinery fails internally and misreports it as
# "cannot import name 'CLIPImageProcessor' from transformers", even though
# CLIPImageProcessor itself is unaffected — the real failure is elsewhere.
# 4.55.4 is verified working with diffusers 0.37.1's Wan pipeline and with
# Qwen-Image-Edit-2511's config format. An open range (">=4.52.0,<5.0")
# will silently pick up a newer, untested transformers release the next
# time this venv is rebuilt and reintroduce this exact crash.
transformers==4.55.4

# WanImageToVideoPipeline (diffusers.pipelines.wan) was added in 0.34.0.
# 0.33.x does not have it and will crash at import time.
# PINNED EXACTLY to the version verified against transformers==4.55.4 above.
# A range here can independently resolve to a diffusers release built
# against a different transformers API and break the same way.
diffusers==0.37.1

safetensors>=0.4.0

# numpy: allow >=1.26 and <2.2. This is compatible with torch 2.8 dev and
# avoids constraining the system install. The venv will install its own copy.
numpy>=1.26,<2.2
